# Task breakdown:
# Stage 1: Move gripper to grasp the wine bottle (keypoint 3) as identified from the image.
# Stage 2: Move gripper with the wine bottle to the top of the wooden cabinet (keypoints 11, 12, 13).

num_stages = 2

### stage 1 guidance (grasp the wine bottle)
def stage1_guidance(keypoints, action_sequence):
    # Guiding the gripper to the wine bottle (keypoint 3)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([3], dtype=torch.long, device=keypoints.device)  # Wine bottle keypoint
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists**2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!

### stage 2 guidance (place the wine bottle on the top of the wooden cabinet)
def stage2_guidance(keypoints, action_sequence):
    # Guiding the gripper to the top of the wooden cabinet (mean of keypoints 11, 12, 13)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    target_indices = torch.tensor([11, 12, 13], dtype=torch.long, device=keypoints.device)  # Cabinet top keypoints
    target_pos = keypoints[target_indices].mean(dim=0)  # Average position for cabinet top
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists**2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!
