# Task breakdown:
# Stage 1: Move gripper to press the stove button (keypoint 20) to turn it on.

num_stages = 1

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Guide the robot to the stove button at keypoint 20.

    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    # Increase weights for actions closer to the end of the sequence
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Keypoint index for the stove button
    target_idx = torch.tensor([20], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Compute distance between action sequence and target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative squared distance

    return reward.mean()  # Return a scalar reward
