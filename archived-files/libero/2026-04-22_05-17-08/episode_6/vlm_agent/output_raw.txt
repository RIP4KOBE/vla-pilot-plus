# Task breakdown:
# Stage 1: Move gripper to grasp the plate (keypoint 5), identified visually as the target plate on the table.
# Stage 2: Move the gripper with the plate to the front of the stove (keypoint 16), identified as the stove burner plate.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to keypoint 5 (plate) to grasp it.
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([5], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Keypoint 5 (plate)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Move the gripper with the plate to keypoint 16 (front of the stove).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([16], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Keypoint 16 (front of the stove)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!
