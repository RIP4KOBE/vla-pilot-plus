# Task breakdown:
# Stage 1: Move gripper to grasp the butter (keypoint 0) which is clearly labeled in the image.
# Stage 2: Move gripper with butter to the basket (any one of keypoints 1, 2, 3, or 4 as they all represent the basket). Since no decision is required for placement, return 0.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to keypoint 0 (butter)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    # Target position for butter (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Calculate distances
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)

    # Weighted reward
    reward = -(all_dists**2 * weights).sum(dim=-1)  # Shape: (B,)
    return reward.mean()  # MUST return scalar!
