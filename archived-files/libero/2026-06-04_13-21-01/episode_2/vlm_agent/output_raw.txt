# Task breakdown:
# Stage 1: Move gripper to grasp the cream cheese (keypoint 0) identified from the image.
# Stage 2: Place the grasped cream cheese into the basket (keypoint 1) identified from the image.

num_stages = 2

### stage 1 guidance (Grasp cream cheese)
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Always use torch.where for weights to avoid memory sharing errors
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Target keypoint for cream cheese
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Cream cheese is keypoint 0
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Use negative squared distance (weighted)
    return reward.mean()  # MUST return scalar!

### stage 2 guidance (Place cream cheese into the basket)
def stage2_guidance(keypoints, action_sequence):
    # Since the task involves placing the object in the basket and no complex decision is necessary,
    # we do not need to create a specific reward function here. Returning 0 is sufficient.
    return torch.tensor(0.0, device=action_sequence.device)  # No additional guidance needed
