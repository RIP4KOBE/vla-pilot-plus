# Task breakdown:
# Stage 1: Move gripper to grasp the salad dressing (keypoint 0) as identified from the image.
# Stage 2: Place the grasped salad dressing into the basket (keypoints 2, 3, 4, 5).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guide gripper to the salad dressing at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Emphasize actions toward the later part of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),  # Higher weight closer to end
        torch.tensor(1.0, device=action_sequence.device)
    )
    
    weights = weights / weights.sum()  # Normalize weights

    # Index for salad dressing (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Position of salad dressing

    # Compute distances along trajectory
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Negative squared distance, weighted

    return reward.mean()  # Return scalar reward
    
### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for placement; return 0 by convention
    return torch.tensor(0.0, device=action_sequence.device)
