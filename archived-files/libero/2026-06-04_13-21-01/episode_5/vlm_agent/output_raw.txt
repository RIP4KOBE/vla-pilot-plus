# Task breakdown:
# Stage 1: Move gripper to grasp ketchup (keypoint 0 or 1, same object visually corresponds to ketchup)
# Stage 2: Move gripper with ketchup to the basket (keypoints 2, 3, 4, 5, 6)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance to move gripper to grasp the ketchup at keypoints 0 and 1
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Keypoints 0 and 1 define the ketchup
    ketchup_indices = torch.tensor([0, 1], dtype=torch.long, device=keypoints.device)
    ketchup_pos = keypoints[ketchup_indices].mean(dim=0)  # Average position of keypoints 0 and 1
    all_dists = torch.norm(action_sequence - ketchup_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision needed; simply return 0 as we let the trained policy handle this stage
    return torch.tensor(0.0, device=action_sequence.device)
