# Task breakdown:
# Stage 1: Move gripper to grasp the ketchup (keypoints 0 and 1, corresponding to 'ketchup_1_main').
# Stage 2: Move gripper with the ketchup to the basket (keypoints 2 through 6, corresponding to 'basket_1_main').

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance for the robot arm to move toward the grasp position of the ketchup (keypoint 0, which is visually aligned at the top of the ketchup bottle).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()
    
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Keypoint for ketchup grasping
    target_pos = keypoints[target_idx][0]  # Position of keypoint 0
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Negative squared distance
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Guidance function for placing the ketchup into the basket (no decision guidance required, so just return 0).
    return torch.tensor(0.0, device=action_sequence.device)
