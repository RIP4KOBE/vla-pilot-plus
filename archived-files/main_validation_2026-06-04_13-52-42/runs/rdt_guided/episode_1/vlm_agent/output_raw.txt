# Task breakdown:
# Stage 1: Move gripper to grasp the alphabet soup (keypoint 0), as identified from the image.
# Stage 2: Move gripper with the alphabet soup to place it in the basket. Return 0 since we do not need a decision at this stage.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # This guidance ensures the robot moves toward the alphabet soup for grasping.
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    # Assign weights for prioritizing later timesteps during trajectory optimization
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Identify the keypoint corresponding to the alphabet soup
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Compute the 3D distance of action sequence to target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    
    # Compute weighted negative squared distance as reward
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision-making needed for placing, so we return 0.
    return torch.tensor(0.0, device=action_sequence.device)
