# Task breakdown:
# Stage 1: Move gripper to grasp the salad dressing (keypoints 0 or 1 - they both correspond to the salad dressing object)
# Stage 2: Move gripper with the grasped salad dressing to the basket (keypoints 2, 3, 4, 5 - these correspond to the basket)
# Note: For stage 2, no decision is needed, so we return 0.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the salad dressing
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Keypoint indices 0 and 1 correspond to the salad dressing
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Choosing keypoint 0
    target_pos = keypoints[target_idx][0]
    
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision needed for placing in the basket; returning 0
    return torch.tensor(0.0, device=action_sequence.device)
