### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp cream cheese (keypoint 0).
    
    T = action_sequence.shape[1]  # Total timesteps
    timesteps = torch.arange(T, device=action_sequence.device)  # [0, 1, ..., T-1]
    final_start = int(0.7 * T)  # Consider final steps more important for precision
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()  # Normalize weights

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Keypoint for cream cheese
    target_pos = keypoints[target_idx][0]  # (3,) coordinates of target keypoint

    # Compute negative squared distance between trajectory and target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted sum over timesteps
    return reward.mean()  # Return scalar reward
