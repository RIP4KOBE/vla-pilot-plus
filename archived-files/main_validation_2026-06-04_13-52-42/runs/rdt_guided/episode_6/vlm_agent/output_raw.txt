# Task breakdown:
# Stage 1: Move gripper to grasp the tomato sauce (keypoint 0).
# Stage 2: Move gripper with the grasped tomato sauce to the basket (keypoints 1, 2, 3, 4, 5).

num_stages = 2

### Stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guide the gripper to the position of the tomato sauce at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Keypoint 0 is for the tomato sauce
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return scalar reward

### Stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision is needed for placing into the basket, so return 0
    return torch.tensor(0.0, device=action_sequence.device)  # Return scalar
