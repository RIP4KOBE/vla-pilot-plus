# Task breakdown:
# Stage 1: Move gripper to grasp milk (keypoint 0).
# Stage 2: Place the milk in the basket (keypoints 1, 2, 3, 4, 5).

num_stages = 2

### Stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance to move gripper to grasp milk at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Milk is keypoint 0
    target_pos = keypoints[target_idx][0]  # Shape (3,)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted summed distances
    return reward.mean()  # Return scalar reward

### Stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # For this task, no need to implement guidance for placing the object in the basket
    return torch.tensor(0.0, device=action_sequence.device)  # Since we don't need to make a decision
