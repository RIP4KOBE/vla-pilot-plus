# Task breakdown:
# Stage 1: Move gripper to grasp the alphabet soup (keypoint 0).
# Stage 2: Place the grasped alphabet soup into the basket (keypoints 1, 2, 3, 4, or 5).
# Note: Stage 2 does not require a guidance function as placing is deterministic.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the alphabet soup (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Alphabet soup keypoint is 0
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Compute distances between action sequence and target
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted sum of negative squared distances
    return reward.mean()  # Return mean scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision needs to be made for placing; return 0
    return torch.tensor(0.0, device=action_sequence.device)
