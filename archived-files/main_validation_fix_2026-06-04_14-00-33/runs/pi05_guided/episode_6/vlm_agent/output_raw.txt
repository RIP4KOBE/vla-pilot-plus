# Task breakdown:
# Stage 1: Move gripper to grasp the tomato sauce (keypoint 0), which is labeled in the image.
# Stage 2: Move gripper with the tomato sauce to the basket (keypoints 1–5). Guidance is not needed for this stage.

num_stages = 2

### Stage 1 guidance: Grasp the tomato sauce (keypoint 0)
def stage1_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Use keypoint 0 for tomato sauce
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Compute distance to target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Scalar reward per batch
    return reward.mean()  # Mean scalar reward for all batches

### Stage 2 guidance: Place the tomato sauce in the basket (keypoints 1–5)
def stage2_guidance(keypoints, action_sequence):
    # No decision is needed for placing; return 0 as per instructions
    return torch.tensor(0.0, device=action_sequence.device)
