# Task breakdown:
# Stage 1: Move gripper to grasp the chocolate pudding (keypoint 0)
# Stage 2: Move gripper with the chocolate pudding to the basket (keypoints 1, 2, 3)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Grasp the chocolate pudding marked as keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    # Higher weight assigned to the end of the trajectory
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Target is keypoint 0 for the chocolate pudding
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]

    # Compute distance from action sequence to the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Negative distance squared for guidance
    return reward.mean()  # Return a scalar

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision-making required for placing in the basket; return 0
    return torch.tensor(0.0, device=action_sequence.device)
