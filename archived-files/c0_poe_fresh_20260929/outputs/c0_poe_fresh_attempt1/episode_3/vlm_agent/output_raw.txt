# Task breakdown:
# Stage 1: Move gripper to grasp the salad dressing (keypoints 0 and 1). I will target the lower body of the bottle (keypoint 0) for a stable grasp.
# Stage 2: Place the salad dressing into the basket (keypoints 2, 3, 4, 5, 6). As per the instructions for basket tasks, the second stage guidance is not strictly required as the policy handles the placement into the container.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Emphasize the final part of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Identifying salad dressing keypoint from image: 0 (lower body)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    
    # Calculate distance for each step in the trajectory
    # Encourage the robot to approach from above by weighting XY distance higher if needed, 
    # but here we use standard Euclidean distance.
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    
    # Reward is negative squared distance (maximized when distance is 0)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # As per CRITICAL instruction: "For scene with pick something into basket, 
    # you do not need to create stage 2 reward function, since we do not need to make decision here, just return 0."
    return torch.tensor(0.0, device=action_sequence.device)
