# Task breakdown:
# Stage 1: Move the gripper to grasp the orange juice (keypoint 0). Keypoint 0 is identified as the main body of the orange juice container.
# Stage 2: Place the grasped orange juice into the basket (represented by keypoints 2, 3, 4, 5, 6). As per instructions for placing into a basket, stage 2 does not require a specific guidance function.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Increase weight for the final part of the trajectory to ensure precise grasping
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Identifying target keypoint 0 for orange juice based on map and image
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    
    # Calculate Euclidean distance between trajectory and target position
    # The robot needs to approach from above, so we give more importance to XY alignment
    xy_dist = torch.norm(action_sequence[..., :2] - target_pos[:2], dim=-1)
    z_dist = torch.abs(action_sequence[..., 2] - target_pos[2])
    
    # Total distance with a slight emphasis on XY alignment for top-down grasp
    total_dist = 2.0 * xy_dist + z_dist
    
    # Reward is negative squared distance (maximize reward = minimize distance)
    reward = -(total_dist ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Per task-specific rules: For scene with pick something into basket, 
    # we do not need to create stage 2 reward function, just return 0.
    return torch.tensor(0.0, device=action_sequence.device)
