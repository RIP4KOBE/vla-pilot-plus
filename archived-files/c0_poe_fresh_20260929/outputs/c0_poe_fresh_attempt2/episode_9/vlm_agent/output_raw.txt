# Task breakdown:
# Stage 1: Move gripper to grasp the chocolate pudding (keypoint 0).
# Stage 2: Place the grasped chocolate pudding into the basket (keypoints 1, 2, 3, 4, 5). As per instructions, placing into a basket does not require specific guidance in stage 2 as the policy handles the spatial placement within the container.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Increase weight for the end of the trajectory to ensure precise reaching
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Identifying chocolate pudding: keypoint 0 from the provided map and image labels
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    
    # Calculate distance between action sequence (gripper pos) and chocolate pudding
    # Give slightly more weight to XY plane for top-down grasp approach
    xy_dist = torch.norm(action_sequence[..., :2] - target_pos[:2], dim=-1)
    z_dist = torch.abs(action_sequence[..., 2] - target_pos[2])
    total_dist = xy_dist * 1.5 + z_dist
    
    reward = -(total_dist ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # As per the instructions: "For scene with pick something into basket, you do not need to create stage 2 reward function, since we do not need to make decision here, just return 0."
    return torch.tensor(0.0, device=action_sequence.device)
