# Task breakdown:
# Stage 1: Move gripper to grasp the bbq sauce (keypoint 0).
# Stage 2: Move the grasped bbq sauce into the basket. The low-level policy knows the basket location, but we can provide a general guidance towards the basket center (mean of keypoints 1, 2, 3, 4, 5). However, per the special instructions for pick into basket tasks, stage 2 can return 0.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    # Focus reward on the end of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # I looked at the image and saw keypoint 0 is the bbq sauce
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    
    # Calculate distance for the whole trajectory
    # As instructed for pick-and-place, we might want to emphasize XY accuracy if picking from above
    xy_dist = torch.norm(action_sequence[..., :2] - target_pos[:2], dim=-1)
    z_dist = torch.abs(action_sequence[..., 2] - target_pos[2])
    
    # Reward is negative distance squared
    # Emphasize XY by doubling its weight in distance calculation
    dist_sq = (2.0 * xy_dist**2 + z_dist**2)
    
    reward = -(dist_sq * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # As per instructions: "For scene with pick something into basket, you do not need to create stage 2 reward function, since we do not need to make decision here, just return 0."
    return torch.tensor(0.0, device=action_sequence.device)
