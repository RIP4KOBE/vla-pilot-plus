# Task breakdown:
# Stage 1: Move gripper to grasp the salad dressing (keypoints 0, 1). Looking at the image, keypoint 1 is at the top/neck of the bottle, which is a good grasp point.
# Stage 2: Place the salad dressing into the basket (keypoints 2, 3, 4, 5, 6). As per instructions for pick-and-place into a basket, no specific guidance is needed for stage 2.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    # Focus on the end of the trajectory for the grasp
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # I looked at the image and saw keypoint 1 is the salad dressing neck/top
    target_idx = torch.tensor([1], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    
    # Calculate Euclidean distance
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    
    # Reward is negative squared distance, prioritized at the end of the sequence
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # As per instructions: "For scene with pick something into basket, you do not need to create stage 2 reward function"
    return torch.tensor(0.0, device=action_sequence.device)
