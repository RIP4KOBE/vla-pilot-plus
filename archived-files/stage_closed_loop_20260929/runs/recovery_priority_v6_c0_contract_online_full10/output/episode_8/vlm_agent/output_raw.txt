# Task breakdown:
# Stage 1: Move the gripper to grasp the milk. The milk is represented by keypoints 0 and 1. 
# According to the instructions, we use the lowest numbered keypoint for the target object, which is keypoint 0.
# Stage 2: Place the grasped milk into the basket. As per the C0 control condition instructions, 
# stage 2 guidance returns a scalar zero tensor.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    # Temporal weights: 3.0 for the final 30% of the trajectory, 1.0 otherwise
    weights = torch.where(timesteps >= final_start, 
                          torch.tensor(3.0, device=action_sequence.device), 
                          torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Target is the milk, using the lowest numbered keypoint (index 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    
    # Calculate isotropic Euclidean XYZ distance
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    
    # Reward is the negative weighted sum of squared distances
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Stage 2 follows the C0 control condition: return a scalar zero tensor
    return torch.tensor(0.0, device=action_sequence.device)
