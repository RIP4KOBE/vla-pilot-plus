# Task breakdown:
# Stage 1: Move gripper to grasp the alphabet soup (keypoint 0).
# Stage 2: Place the grasped alphabet soup into the basket (keypoint 1).
# Note: Stage 2 does not need a reward function, just return 0.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Guiding the robot to the alphabet soup at keypoint 0.
    T = action_sequence.shape[1]  # Number of timesteps
    timesteps = torch.arange(T, device=action_sequence.device)  # Timesteps array
    final_start = int(0.7 * T)  # Start of the final weight range
    # Assign higher weights to the latter part of the sequence
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()  # Normalize weights

    # Keypoint for the target object (alphabet soup)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Calculate the distance between each point in the action sequence and the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)

    # Reward is higher when the action sequence gets closer to the target
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative squared distance
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision is needed for placement; return 0.
    return torch.tensor(0.0, device=action_sequence.device)
