def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Guidance for grasping the cream cheese (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Target position for the cream cheese
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Keypoint index for cream cheese
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Calculate distances between the trajectory and the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)

    # Apply weighted distance for reward
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Returning the mean reward across batches as a scalar
