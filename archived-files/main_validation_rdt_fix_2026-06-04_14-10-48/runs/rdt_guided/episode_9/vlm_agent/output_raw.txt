# Task breakdown:
# Stage 1: Move gripper to grasp chocolate pudding (keypoint 0).
# Stage 2: Place the chocolate pudding in the basket (keypoints 1, 2, 3, 4). No decision is needed here, just return 0.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move the gripper to grasp chocolate pudding (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Assign higher weight to the final part of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Chocolate pudding index
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return scalar guidance

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision needed for placing in the basket
    return torch.tensor(0.0, device=action_sequence.device)  # Directly return 0
