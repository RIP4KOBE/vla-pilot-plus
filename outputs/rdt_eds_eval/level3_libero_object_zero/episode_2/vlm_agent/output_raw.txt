# Task breakdown:
# Stage 1: Move gripper to grasp the "cream cheese" (keypoint 0) which is visible in the scene.
# Stage 2: Move gripper with the "cream cheese" to the "basket" (keypoints 1, 2, 3, 4). No guidance needed for stage 2.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance to move gripper toward the "cream cheese" (keypoint 0).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    # Weight the last 30% of the trajectory more heavily
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    # Target position for "cream cheese" (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Calculate squared distances
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Shape: (B,)
    
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for this stage (policy knows where to place once the object is grasped).
    return torch.tensor(0.0, device=action_sequence.device)  # Return 0
