# Task breakdown:
# Stage 1: Move gripper to grasp the ketchup, which corresponds to keypoint 0
# Stage 2: Place the grasped ketchup in the basket, which corresponds to the region of keypoints 2, 3, 4, 5, 6. (No additional decision-making needed for stage 2)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guide the robot to move close to the ketchup for grasping
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    weights = torch.where(timesteps >= final_start, 
                          torch.tensor(3.0, device=action_sequence.device), 
                          torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Ketchup corresponds to keypoint 0
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative squared distance
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No need to provide a specific reward for placing in the basket, as this is handled by the diffusion model
    return torch.tensor(0.0, device=action_sequence.device)
