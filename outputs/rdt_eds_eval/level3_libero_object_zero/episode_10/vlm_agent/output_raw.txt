# Task breakdown:
# Stage 1: Move gripper to grasp the orange juice (keypoint indices 0 or 1). I can see from the image that the orange juice corresponds to these keypoints.
# Stage 2: Move the gripper with the orange juice to the basket (keypoints 2, 3, 4, 5). No action guidance is needed for stage 2.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Target orange juice keypoints (choosing the first keypoint, index 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]

    # Compute distance reward to target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No specific motion guidance is required for the placement task (basket).
    # The policy knows how to execute the placement in stage 2. Return 0.
    return torch.tensor(0.0, device=action_sequence.device)
