# Task breakdown:
# Stage 1: Move gripper to grasp the butter (keypoint 0) which I identified visually as "butter_1_main".
# Stage 2: Move gripper with the butter to place it in the basket (keypoints 1, 2, 3, 4).
# Note: Stage 2 does not require explicit guidance, as the basket has a single location.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # This stage guides the robot to move to the butter for grasping.
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)  # Higher weight near the end of the sequence
    
    # Weights for the trajectory
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Target position (butter keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape (3,)
    
    # Compute distance reward
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted reward
    return reward.mean()  # Must return scalar

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for this stage (policy knows how to place in basket).
    return torch.tensor(0.0, device=action_sequence.device)
