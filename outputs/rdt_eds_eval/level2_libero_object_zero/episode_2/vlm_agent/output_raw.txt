# Task breakdown:
# Stage 1: Move gripper to grasp alphabet soup (keypoint 0)
# Stage 2: Move gripper with grasped alphabet soup to basket (keypoint 1, or any part of basket)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp alphabet soup (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    # Apply higher importance to time steps closer to the end
    final_start = int(0.7 * T) 
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()
    
    # Keypoint index for alphabet soup
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    # Compute distances and apply the weighted sum
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return a scalar value

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No additional decision-making is needed to place the object in the basket
    return torch.tensor(0.0, device=action_sequence.device)  # Return zero reward since policy handles placement
