# Task breakdown:
# Stage 1: Move gripper to grasp the butter, which corresponds to keypoint 8.
# Stage 2: Move gripper with the butter to the basket, which corresponds to keypoints 2, 3, 4, and 5.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move the gripper to grasp the butter (keypoint 8).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()
    
    # Select keypoint associated with "butter_1_main".
    target_idx = torch.tensor([8], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]

    # Compute distance and reward.
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    
    return reward.mean()  # Return scalar reward.

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for stage 2 as the task specifies that the basket is the only destination.
    return torch.tensor(0.0, device=action_sequence.device)
