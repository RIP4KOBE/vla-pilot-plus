def stage1_guidance(keypoints, action_sequence):
    # Move gripper to the BBQ sauce at keypoint 11
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Assign higher weights to the later part of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, 
                          torch.tensor(3.0, device=action_sequence.device), 
                          torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Target position is defined by keypoint 11
    target_idx = torch.tensor([11], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape (3,)
    
    # Compute distance between action sequence and target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative distance squared
    return reward.mean()  # MUST return a scalar
