# Task breakdown:
# Stage 1: Move the gripper to grasp the ketchup bottle (keypoint 6)
# Stage 2: Move the gripper with the ketchup bottle to the basket (no guidance required for placement)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Give higher weights to the final part of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # The ketchup bottle corresponds to keypoint 6
    target_idx = torch.tensor([6], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]

    # Calculate distance from each point in the trajectory to the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)

    # Reward is higher (closer to zero) as the distance decreases
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return scalar reward value

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No choice about where to place the ketchup (basket is predetermined)
    # Return a default scalar value of 0
    return torch.tensor(0.0, device=action_sequence.device)
