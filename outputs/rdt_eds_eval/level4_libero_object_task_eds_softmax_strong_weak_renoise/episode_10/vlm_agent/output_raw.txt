# Task breakdown:
# Stage 1: Move gripper to grasp the chocolate pudding (keypoint 6)
# Stage 2: Move gripper with the chocolate pudding to the basket (keypoint 1). 
#          No guidance function is required for this stage since no decision-making is involved.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance for moving the gripper to grasp the chocolate pudding (keypoint 6)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([6], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Get the position of the chocolate pudding
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision needed for placing in the basket; return 0 as guidance function
    return torch.tensor(0.0, device=action_sequence.device)
