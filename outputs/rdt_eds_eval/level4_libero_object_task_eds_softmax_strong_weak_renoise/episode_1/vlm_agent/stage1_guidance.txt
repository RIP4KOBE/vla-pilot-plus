def stage1_guidance(keypoints, action_sequence):
    # Guidance to grasp the cream cheese (keypoint 7)
    
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    # Weighting for emphasizing final timesteps
    weights = torch.where(timesteps >= final_start, 
                          torch.tensor(3.0, device=action_sequence.device), 
                          torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Identify keypoint for cream cheese
    cream_cheese_idx = torch.tensor([7], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[cream_cheese_idx][0]  # Shape: (3,)

    # Calculate the distance to cream cheese from the trajectory
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)

    # Compute reward (negative squared distances weighted by timestep weights)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()
