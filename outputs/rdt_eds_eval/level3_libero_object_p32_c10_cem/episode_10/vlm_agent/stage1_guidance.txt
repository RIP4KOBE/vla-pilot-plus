def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Guidance for grasping orange juice (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)  # Focus on the last 30% of the trajectory
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()  # Normalize weights

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Keypoint for orange juice
    target_pos = keypoints[target_idx][0]  # Extract the position of the keypoint
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Compute distances to the target
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted distance for guidance
    return reward.mean()  # Return scalar reward
