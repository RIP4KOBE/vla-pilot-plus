# Task breakdown:
# Stage 1: Move the gripper to grasp the cream cheese (keypoint 0) which is clearly visible in front of the robot
# Stage 2: Move the gripper with the cream cheese to the basket (keypoints 1, 2, 3, 4, 5)

num_stages = 2

### stage 1 guidance - Move gripper to cream cheese (keypoint 0)
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Assign weights to prioritize the late part of the trajectory
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Target position for the cream cheese (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Calculate distances to the target position across the entire trajectory
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative squared distance

    return reward.mean()  # Return scalar reward

### stage 2 guidance - Move the cream cheese into the basket (keypoints 1, 2, 3, 4, 5)
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for the placement; return 0
    return torch.tensor(0.0, device=action_sequence.device)
