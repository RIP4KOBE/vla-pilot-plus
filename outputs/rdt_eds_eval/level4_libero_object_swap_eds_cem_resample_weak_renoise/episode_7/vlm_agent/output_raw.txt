# Task breakdown:
# Stage 1: Move gripper to grasp the butter (keypoint 0)
# Stage 2: Move gripper (carrying the butter) to place it in the basket (keypoints 1, 2, 3, 4). No decision is required for this stage, so we return 0.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Ensure the guidance code is fully self-contained
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Keypoint for butter
    target_pos = keypoints[target_idx][0]  # Get the position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return a scalar value

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision-making is required for this stage; the policy handles placing in the basket. 
    # Simply return 0.
    return torch.tensor(0.0, device=action_sequence.device)
