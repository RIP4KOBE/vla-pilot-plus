# Task breakdown:
# Stage 1: Move the gripper to grasp the ketchup (keypoint 0), clearly identified in the image.
# Stage 2: Place the ketchup in the basket (keypoints 2, 3, 4, and 5); no guidance needed for this stage.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move the gripper to grasp the ketchup (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)

    # Prioritize the final timesteps more during trajectory planning
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Target position for ketchup (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape (3,)

    # Compute 3D distance between every timestep in the trajectory and the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)

    # Minimize the squared distances weighted over the trajectory
    reward = -(all_dists ** 2 * weights).sum(dim=1)

    return reward.mean()  # MUST return a scalar!

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No position decision is required for placing ketchup in basket.
    # Simply return 0 as no guidance is needed.
    return torch.tensor(0.0, device=action_sequence.device)
