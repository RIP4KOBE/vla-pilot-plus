def stage1_guidance(keypoints, action_sequence):
    # Move to grasp the milk
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Select keypoint for the milk
    milk_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Using keypoint 0 for milk
    target_pos = keypoints[milk_idx][0]
    
    # Calculate reward
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return scalar reward
