# Task breakdown:
# Stage 1: Move the gripper to grasp the alphabet soup (keypoint 0). 
# Stage 2: Move the gripper with the alphabet soup to place it into the basket (keypoints 1, 2, 3, 4).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move the gripper to grasp the alphabet soup, identified as keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    soup_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    soup_pos = keypoints[soup_idx][0]
    all_dists = torch.norm(action_sequence - soup_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No keypoint selection is required for placing into the basket
    # Just return 0, as the policy knows how to place in the basket
    return torch.tensor(0.0, device=action_sequence.device)
