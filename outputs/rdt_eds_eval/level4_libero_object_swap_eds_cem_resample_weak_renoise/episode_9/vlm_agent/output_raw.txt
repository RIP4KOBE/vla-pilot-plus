# Task breakdown:
# Stage 1: Move gripper to grasp the chocolate pudding (keypoint 0), which is visually identified in the scene.
# Stage 2: Move the gripper holding the chocolate pudding to the basket (keypoint 1). No guidance function needed for stage 2 as placement decisions are handled by the policy.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Target is the chocolate pudding, identified as keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Emphasize later timesteps for reward computation
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Specify target position for chocolate pudding (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]

    # Compute distance to target
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance function required for placement in the basket (keypoint 1)
    return torch.tensor(0.0, device=action_sequence.device)  # Return scalar 0
