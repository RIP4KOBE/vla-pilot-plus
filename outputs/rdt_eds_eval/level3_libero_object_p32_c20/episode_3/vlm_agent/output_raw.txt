# Task breakdown:
# Stage 1: Move gripper to grasp the salad dressing (keypoint 0) which is identified in the image.
# Stage 2: Move gripper with the salad dressing to the basket (keypoints 2, 3, 4, 5). No guidance function needed here as the placement location is explicit.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the salad dressing
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Salad dressing keypoint is clearly identified as keypoint 0
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Ensure shape (3,)

    # Calculate distance to aim for the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return a scalar reward value

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No need to guide for placement in the basket; simply return 0
    return torch.tensor(0.0, device=action_sequence.device)
