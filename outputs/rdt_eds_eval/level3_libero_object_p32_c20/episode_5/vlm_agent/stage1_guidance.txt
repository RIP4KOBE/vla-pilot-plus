def stage1_guidance(keypoints, action_sequence):
    # ALL code inside this function - no external helpers!
    T = action_sequence.shape[1]  # Number of timesteps in the action sequence
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    # Weights to encourage precision near the end of the trajectory
    weights = torch.where(timesteps >= final_start, 
                          torch.tensor(3.0, device=action_sequence.device), 
                          torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()  # Normalize weights

    # Target position for ketchup (keypoint 1)
    target_idx = torch.tensor([1], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape (3,)

    # Calculate negative squared distance to target
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted reward per batch
    return reward.mean()  # Return scalar
