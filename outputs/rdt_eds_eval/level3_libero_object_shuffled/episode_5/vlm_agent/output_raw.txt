# Task breakdown:
# Stage 1: Move gripper to grasp the ketchup bottle (keypoint 0).
# Stage 2: Move gripper with grasped ketchup to above the basket (no decision needed here, return 0).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guide the gripper to move to keypoint 0 (ketchup bottle).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)  # Focus on the final 30% of the trajectory
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Keypoint 0 corresponds to ketchup
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Compute distances
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted sum of distances
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision needed for placing the ketchup above the basket; just return 0.
    return torch.tensor(0.0, device=action_sequence.device)
