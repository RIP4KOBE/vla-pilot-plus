# Task breakdown:
# Stage 1: Move gripper to grasp the orange juice (keypoint 0 or 1 - visually identified as the same object)
# Stage 2: Place the orange juice into the basket (keypoints 2, 3, 4, 5, or 6 - visually identified as part of the basket)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance to move the gripper to grasp the orange juice
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    # Higher reward for later timesteps to focus trajectory completion near the target
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Keypoint 0 identified visually as orange juice
    target_pos = keypoints[target_idx][0]

    # Compute 3D distance between the trajectory and the target
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Negative distance squared times weights
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision needed for placing the orange juice in the basket
    # The policy uses its pre-learned motion to complete this stage
    return torch.tensor(0.0, device=action_sequence.device)  # Always return 0 for stage 2
