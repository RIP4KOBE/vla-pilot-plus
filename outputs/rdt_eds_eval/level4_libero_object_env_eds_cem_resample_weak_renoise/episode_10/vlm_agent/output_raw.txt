# Task breakdown:
# Stage 1: Move gripper to grasp the "orange juice". From the image, I can see that keypoint 6 is associated with the orange juice.
# Stage 2: Move gripper with the grasped object into the basket. From the image, I can see that keypoint 0 corresponds to the basket.
# Note: No guidance is required for the second stage since the policy knows how to place objects in the basket.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Targeting keypoint 6 for the orange juice
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()  # Normalize weights

    # Keypoint index for "orange juice"
    target_idx = torch.tensor([6], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Coordinates of keypoint 6

    # Compute rewards based on proximity to target
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Distance to target
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted sum of negative squared distances
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # As per instructions, no guidance is needed for placing objects into the basket
    return torch.tensor(0.0, device=action_sequence.device)  # Return a constant reward of 0
