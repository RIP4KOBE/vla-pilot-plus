def stage1_guidance(keypoints, action_sequence):
    # Move to grasp ketchup (keypoint 3)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    # Assign higher weights to the later parts of the trajectory
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()
    
    # Select the ketchup keypoint (keypoint 3)
    ketchup_idx = torch.tensor([3], dtype=torch.long, device=keypoints.device)
    ketchup_pos = keypoints[ketchup_idx][0]
    
    # Compute distance to ketchup keypoint
    all_dists = torch.norm(action_sequence - ketchup_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()
