# Stage 1: Move gripper to grasp the milk (hardcoded to keypoint 7 based on image observation)
# Stage 2: Move gripper with the milk to place it in the basket (no decision needed for placement, return 0)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Grasp the milk (keypoint 7)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Assign higher weight to positions near the final 30% of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    # Get the position of the milk (keypoint 7)
    milk_idx = torch.tensor([7], dtype=torch.long, device=keypoints.device)
    milk_pos = keypoints[milk_idx][0]  # Shape (3,)
    
    # Compute distances between trajectory points and target position
    all_dists = torch.norm(action_sequence - milk_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return a scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Stage 2: Move the milk to the basket (no decision needed; return 0)
