# Task breakdown:
# Stage 1: Move the gripper to grasp the butter (keypoint 0) which is visually identified on the scene.
# Stage 2: Place the grasped butter inside the basket (keypoints 1, 2, 3, 4, 5). No guidance needed for this stage.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # This function guides the policy to move the gripper toward the butter (keypoint 0).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Butter keypoint
    target_pos = keypoints[target_idx][0]  # Shape (3,)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative distance
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No specific decision-making is required for placing the object in the basket.
    return 0.0  # Return a constant reward value for this stage
