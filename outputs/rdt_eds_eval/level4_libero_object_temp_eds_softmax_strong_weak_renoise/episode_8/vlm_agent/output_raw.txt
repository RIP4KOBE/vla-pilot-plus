### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move the gripper to grasp the milk (keypoint 4)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([4], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Position of the milk
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Distance from the trajectory to milk
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Negative for maximizing proximity to trajectory
    return reward.mean()  # Return scalar reward
