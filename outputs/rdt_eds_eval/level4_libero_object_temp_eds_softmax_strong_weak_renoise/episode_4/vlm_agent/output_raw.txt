# Task breakdown:
# Stage 1: Move gripper to grasp the bbq sauce (keypoint 0) which I identified as the correct object to pick up from the image.
# Stage 2: Move gripper with bbq sauce to the basket (keypoints 2, 3, 4). No guidance needed for this stage; return 0.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Code for guiding the robot to grasp the bbq sauce located at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # bbq sauce keypoint
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision needed for placing in the basket
    return torch.tensor(0.0, device=action_sequence.device)
