def stage1_guidance(keypoints, action_sequence):
    # Move the gripper to grasp the alphabet soup.
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    # Apply a weighted focus towards the later part of the trajectory.
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, 3.0, 1.0)
    weights = weights / weights.sum()

    # Identify the keypoint for the alphabet soup (keypoint 0).
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Get the 3D coordinates of keypoint 0.

    # Compute the distance between the action sequence and the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists**2 * weights).sum(dim=1)  # Squared distance with weights applied.
    return reward.mean()  # Return scalar reward.
