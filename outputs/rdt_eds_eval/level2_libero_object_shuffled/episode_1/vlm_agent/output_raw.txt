# Task breakdown:
# Stage 1: Move gripper to grasp the alphabet soup (keypoint 0)
# Stage 2: Move gripper with the alphabet soup to the basket (keypoint 1 - no reward function needed for this stage)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    T = action_sequence.shape[1]  # Get the number of time steps
    timesteps = torch.arange(T, device=action_sequence.device)  # Create an index tensor for timesteps
    final_start = int(0.7 * T)  # Final phase: last 30% of the timesteps
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),  # Higher weight for final 30% timesteps
        torch.tensor(1.0, device=action_sequence.device),  # Normal weight
    )
    weights = weights / weights.sum()  # Normalize weights

    # Select the keypoint corresponding to the alphabet soup (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Calculate the distance between action sequence and the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists**2 * weights).sum(dim=1)  # Weighted negative squared distances
    return reward.mean()  # Return the scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision-making is required for this stage. Returning 0 is sufficient.
    return torch.tensor(0.0, device=action_sequence.device)
