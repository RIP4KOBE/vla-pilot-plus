# Task breakdown:
# Stage 1: Move gripper to grasp the alphabet soup (keypoint 0), as identified in the image.
# Stage 2: Place the alphabet soup into the basket (keypoints 1, 2, 3, 4, 5). 
# Since no decision is needed for the placement location within the basket, we will return 0 for the stage 2 reward function.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to the alphabet soup (keypoint 0)
    T = action_sequence.shape[1]  # Number of timesteps
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)  # Last 30% of the trajectory
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()  # Normalize weights

    # Target position is the keypoint corresponding to the alphabet soup (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Calculate distances between trajectory and target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative squared distance
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision is required here since the basket location is predefined.
    # Thus, we return 0 as the reward for this stage.
    return torch.tensor(0.0, device=action_sequence.device)
