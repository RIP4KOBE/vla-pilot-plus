# Task breakdown:
# Stage 1: Move gripper to grasp the alphabet soup can (keypoint 0).
# Stage 2: Move gripper with the grasped alphabet soup to place it in the basket (keypoints 1, 2, 3, 4, 5).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guide gripper to grasp the alphabet soup can at keypoint 0.
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Index for alphabet soup can
    target_pos = keypoints[target_idx][0]  # Position of the alphabet soup can
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No specific decision needed for placement, return 0.
    return torch.tensor(0.0, device=action_sequence.device)
