# Task breakdown:
# Stage 1: Move gripper to grasp chocolate pudding (keypoint 0).
# Stage 2: Place the chocolate pudding into the basket (no decision-making required; just return 0).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance function for grasping chocolate pudding at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Chocolate pudding keypoint
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)

    reward = -(all_dists**2 * weights).sum(dim=1)  # Reward encourages minimizing distance to target
    return reward.mean()  # Return scalar reward

def stage2_guidance(keypoints, action_sequence):
    # No decision required for placing the object in the basket. Policy will handle it.
