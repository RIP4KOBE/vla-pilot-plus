# Task breakdown:
# Stage 1: Move gripper to grasp cream cheese (keypoint 0)
# Stage 2: Move gripper with cream cheese to the basket (keypoints 1, 2, 3, 4, 5)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance to move the gripper to grasp the cream cheese (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)

    # Emphasize guidance towards the end of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, 
                          torch.tensor(3.0, device=action_sequence.device), 
                          torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Keypoint index for cream cheese is 0
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Get 3D position of the cream cheese (keypoint 0)

    # Compute 3D distance from action sequence to target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Maximize proximity to target
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # THIS STAGE DOES NOT REQUIRE DECISION GUIDANCE.
    # No decision needs to be made for placing cream cheese in the basket.
