# Task breakdown:
# Stage 1: Move the gripper to grasp the milk (keypoints 0 and 1).
# Stage 2: Move the gripper with the grasped milk into the basket (keypoints 2, 3, 4, 5, and 6).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Guidance for moving to the milk before grasping
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    milk_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    milk_pos = keypoints[milk_idx][0]  # Position of the milk keypoint
    all_dists = torch.norm(action_sequence - milk_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # For placing into the basket, guidance is not needed based on task constraints.
    # Simply return 0, as the policy knows how to place in the basket.
    return torch.tensor(0.0, device=action_sequence.device)
