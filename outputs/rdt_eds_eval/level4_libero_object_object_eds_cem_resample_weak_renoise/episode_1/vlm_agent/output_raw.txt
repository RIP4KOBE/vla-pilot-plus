# Task breakdown:
# Stage 1: Grasp the alphabet soup can (keypoint 0, which I identified as the top main keypoint for the alphabet soup).
# Stage 2: Place the alphabet soup can into the basket (keypoints 2, 3, 4, 5, 6 define the basket). [Guidance not needed as the policy already handles placement.]

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move the gripper to grasp the alphabet soup can (keypoint 0).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    # The target position is the keypoint 0 for the alphabet soup can.
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]

    # Compute distances from the action sequence to the target position.
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return a scalar value.

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision needed for placing into the basket; return 0 as per instructions.
    return torch.tensor(0.0, device=action_sequence.device)
