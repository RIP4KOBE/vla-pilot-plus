def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the cream cheese (keypoint 0)
    T = action_sequence.shape[1]  # Number of time steps
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Assign higher weight to the final segment of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()  # Normalize weights for stability

    # Identify the target keypoint position for 'red_cream_cheese_1_main'
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Compute distances and reward
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted sum across timesteps -> Shape: (B,)
    return reward.mean()  # Return scalar mean reward
