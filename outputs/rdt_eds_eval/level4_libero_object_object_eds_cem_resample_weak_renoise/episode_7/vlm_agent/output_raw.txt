# Task breakdown:
# Stage 1: Move gripper to grasp the butter (keypoint 0), as identified in the image.
# Stage 2: Move gripper with the butter to the basket (keypoints 1, 2, 3, and 4). No specific guidance is required for this placement stage.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Moving gripper to grasp the butter (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    # Applying higher weight to final trajectory steps
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Target: Butter (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]

    # Calculate distances and reward
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists**2 * weights).sum(dim=1)
    return reward.mean()  # Return scalar!

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Moving the butter to the basket (target location is already specified by policy)
    # No additional decision is needed for this stage, so we return 0.
    return torch.tensor(0.0, device=action_sequence.device)
