# Task breakdown:
# Stage 1: Move gripper to grasp the yellow tomato sauce (keypoint 0).
# Stage 2: Move gripper with the yellow tomato sauce to the basket (keypoint 1). Since there is no decision-making required, we return 0.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the yellow tomato sauce (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)

    # Apply higher weights towards the end of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Identify the position of the yellow tomato sauce (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape (3,)

    # Compute distance and reward
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return scalar
