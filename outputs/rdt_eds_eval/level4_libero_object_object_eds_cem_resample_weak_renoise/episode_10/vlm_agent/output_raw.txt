# Task breakdown:
# Stage 1: Move gripper to grasp the orange juice (keypoint 1) identified by visual inspection.
# Stage 2: Move gripper with the grasped orange juice toward the basket (keypoints 2, 3, 4, 5). 
#           No guidance function required—policy handles placing.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Code to guide the gripper to the orange juice (keypoint 1)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()
    
    target_idx = torch.tensor([1], dtype=torch.long, device=keypoints.device)  # Orange juice keypoint (1)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative square distances
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No action needed for placing in the basket (policy knows how to place correctly)
    return torch.tensor(0.0, device=action_sequence.device)  # Always return 0 for stage 2
