# Task breakdown:
# Stage 1: Move gripper to grasp the tomato sauce (keypoint 11)
# Stage 2: Move gripper with the tomato sauce to the basket (keypoints 2-5)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the tomato sauce
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([11], dtype=torch.long, device=keypoints.device)  # Keypoint for tomato sauce
    target_pos = keypoints[target_idx][0]  # Position of the tomato sauce keypoint
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Calculate distances
    reward = -(all_dists**2 * weights).sum(dim=1)  # Negative squared distance
    return reward.mean()  # MUST return scalar!

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Move the tomato sauce to the basket; no decision-making is required here
    return torch.tensor(0.0, device=action_sequence.device)  # No guidance required
