# Task breakdown:
# Stage 1: Move gripper to grasp orange juice (keypoint 6)
# Stage 2: Move gripper with orange juice to the basket (keypoints 1, 2, 3, 4)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guiding the robot to the orange juice (keypoint 6)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    # Keypoint 6 corresponds to orange juice
    target_idx = torch.tensor([6], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Compute distances between action sequence and target
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists**2 * weights).sum(dim=1)  # Guided to minimize distance

    return reward.mean()  # Must return a scalar

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for the basket placement stage
    return torch.tensor(0.0, device=action_sequence.device)
