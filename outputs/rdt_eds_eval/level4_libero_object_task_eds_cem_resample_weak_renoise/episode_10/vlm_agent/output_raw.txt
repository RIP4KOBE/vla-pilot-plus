# Task breakdown:
# Stage 1: Move gripper to grasp chocolate pudding (keypoint 6) by approaching it.
# Stage 2: Place the grasped chocolate pudding into the basket (keypoint 1).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Determine the target keypoint for chocolate pudding
    target_idx = torch.tensor([6], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Select chocolate pudding's keypoint (keypoint 6)
    
    # Calculate the weighted distance to guide the gripper to the chocolate pudding
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)  # Assign higher weight near the later part of the trajectory
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Compute distance to keypoint 6
    reward = -(all_dists**2 * weights).sum(dim=1)  # Reward is maximum (less negative) when close to the chocolate pudding
    return reward.mean()  # Return a scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No specific decision is needed here for placement in the basket as per task rules.
    return torch.tensor(0.0, device=action_sequence.device)
