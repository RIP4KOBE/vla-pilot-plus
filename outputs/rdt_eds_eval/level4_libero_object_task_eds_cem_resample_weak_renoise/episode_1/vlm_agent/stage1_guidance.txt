def stage1_guidance(keypoints, action_sequence):
    # Guidance to move gripper to the cream cheese (keypoint 7).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Assign higher weight to the last 30% of the trajectory to focus on the final position.
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Target position is the keypoint for cream cheese.
    target_idx = torch.tensor([7], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Calculate negative squared distance to encourage moving close to the cream cheese.
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!
