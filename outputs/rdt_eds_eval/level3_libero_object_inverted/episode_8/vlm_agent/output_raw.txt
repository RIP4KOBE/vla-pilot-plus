# Task breakdown:
# Stage 1: Move gripper to grasp the milk (keypoint 0)
# Stage 2: Move gripper with milk to the basket (keypoints 1, 2, 3, 4, 5) 
#           - No decision-making is required here for placement, return 0.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Grasp the milk (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    # Target: milk at keypoint 0
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Keypoint index for milk
    target_pos = keypoints[target_idx][0]  # Shape (3,) for milk position

    # Compute weighted negative squared 3D distances
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists**2 * weights).sum(dim=1)  # Weighted reward
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision-making needed for placing in the basket
    # Simply return 0
    return torch.tensor(0.0, device=action_sequence.device)
