# Task breakdown:
# Stage 1: Move gripper to grasp cream cheese (keypoint 0)
# Stage 2: Move gripper with cream cheese to basket (keypoints 1, 2, 3, 4)
# Note: For stage 2, no specific guidance is needed, return 0 as instructed.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Encourage the robot to move close to the cream cheese (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Weight the timesteps towards the end
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()  # Normalization

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Position of the cream cheese (keypoint 0)

    # Calculate distances and weighted rewards
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)

    return reward.mean()  # Return as scalar

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for placing the item into the basket
    return torch.tensor(0.0, device=action_sequence.device)
