def stage1_guidance(keypoints, action_sequence):
    # Guidance for moving the gripper to grasp the orange juice at keypoint 1
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    # Assign higher weights for latter timesteps of the trajectory
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Target position for orange juice (keypoint 1)
    target_idx = torch.tensor([1], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]

    # Calculate the distance between the action sequence points and the target keypoint
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Negative squared distance weighted by weights
    
    return reward.mean()  # Return scalar reward
