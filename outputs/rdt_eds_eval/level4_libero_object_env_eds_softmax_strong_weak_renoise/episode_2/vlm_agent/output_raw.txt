# Task breakdown:
# Stage 1: Move the gripper to grasp the cream cheese at keypoint 0
# Stage 2: Move the gripper (with the cream cheese) to the basket (keypoints 1-4)

num_stages = 2

### stage 1 guidance: Grasp the cream cheese (keypoint 0)
def stage1_guidance(keypoints, action_sequence):
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()  # Normalize weights

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Select cream cheese position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar reward

### stage 2 guidance: Place the cream cheese into the basket
def stage2_guidance(keypoints, action_sequence):
    # No decision needed here, as there's only one basket
    return torch.tensor(0.0, device=keypoints.device)  # Return 0 as per instruction
