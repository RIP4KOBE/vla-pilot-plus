# Task breakdown:
# Stage 1: Move gripper to grasp salad dressing (keypoint 0), visually identified as "salad_dressing_1_main"
# Stage 2: Place the salad dressing in the basket (keypoints 2, 3, 4, 5), no decision guidance needed here (policy handles)

num_stages = 2

### stage 1 guidance (grasp salad dressing)
def stage1_guidance(keypoints, action_sequence):
    # Hardcoded index of the salad dressing keypoint (0) based on image
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Extract coordinate of the salad dressing

    # Weighted time distribution toward end (last 30% of trajectory)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()  # Normalize weights

    # Negative squared distance to encourage closer trajectory to the grasp position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)

    return reward.mean()  # Return scalar reward

### stage 2 guidance (place salad dressing in the basket)
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for placement—policy handles directly
    return torch.tensor(0.0, device=action_sequence.device)
