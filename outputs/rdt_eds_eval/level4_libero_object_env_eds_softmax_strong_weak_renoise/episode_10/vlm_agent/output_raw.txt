# Task breakdown:
# Stage 1: Move gripper to grasp the orange juice (keypoint 6) which is identifiable from the keypoint map and image.
# Stage 2: Move the gripper with the grasped orange juice to the basket (keypoint 0). In this case, no decision-making is required for placement.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the orange juice (keypoint 6)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Assign weights to prioritize the end of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Target position is keypoint 6 (orange juice)
    target_idx = torch.tensor([6], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]

    # Calculate distance from action sequence to target position (keypoint 6)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for placement in the basket (keypoint 0) since policy handles placement
    return torch.tensor(0.0, device=action_sequence.device)
