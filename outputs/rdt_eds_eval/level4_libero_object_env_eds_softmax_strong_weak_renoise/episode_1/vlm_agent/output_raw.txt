# Task breakdown:
# Stage 1: Move gripper to grasp the alphabet_soup can (keypoint 0).
# Stage 2: Place the alphabet_soup can into the basket (keypoints 1, 2, 3, 4).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # This function guides the policy to move the gripper to grasp the alphabet_soup can (keypoint 0).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Applying higher weights to the last 30% of the trajectory to ensure final accuracy.
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()
    
    # Hardcoded index for alphabet_soup can.
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Compute distances and rewards.
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted reward based on distance.
    return reward.mean()  # Return a scalar reward.

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # For placing the can in the basket, we do not need decision-making, so we return a constant reward.
    return torch.tensor(0.0, device=action_sequence.device)
