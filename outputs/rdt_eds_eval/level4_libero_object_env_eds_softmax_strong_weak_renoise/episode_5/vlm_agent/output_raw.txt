# Task breakdown:
# Stage 1: Move gripper to grasp the ketchup (keypoint 3)
# Stage 2: Place the ketchup into the basket (keypoints 0, 1, 2) - No additional guidance function needed here

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Approach the ketchup bottle for grasping (keypoint index 3)
    
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)  # Focus on the final part of the trajectory
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([3], dtype=torch.long, device=keypoints.device)  # Keypoint for ketchup
    target_pos = keypoints[target_idx][0]  # Extract position of ketchup

    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Compute distances to ketchup
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Reward is negative distance, weighted
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for placing in the basket
    return torch.tensor(0.0, device=action_sequence.device)
