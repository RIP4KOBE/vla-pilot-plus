# Task breakdown:
# Stage 1: Move gripper to grasp the alphabet soup (keypoint 0) which I identified in the image.
# Stage 2: Move gripper with the grasped alphabet soup to the basket (keypoint 1 - no decision needed here, return 0).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move the gripper to the alphabet soup (keypoint 0).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    # Weight future timesteps more heavily.
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Target position for the alphabet soup (keypoint 0).
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    
    # Calculate 3D distance from each timestep to the target position.
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return a scalar reward.

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision is needed for placing the object into the basket (keypoint 1).
    return torch.tensor(0.0, device=action_sequence.device).mean()
