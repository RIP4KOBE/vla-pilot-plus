# Task breakdown:
# Stage 1: Move the gripper to grasp the milk, which corresponds to keypoint 0.
# Stage 2: Move the gripper with the milk to the basket, represented by keypoints [1, 2, 3, 4, 5].
# Note: For stage 2, no specific guidance function is needed as the policy can handle placing in the basket.

num_stages = 2

### Stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move the gripper to grasp the milk at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device),
                          torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    milk_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[milk_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

    # No specific guidance needed for placing in the basket, return 0.
