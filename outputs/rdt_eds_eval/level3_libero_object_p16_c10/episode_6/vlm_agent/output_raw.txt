# Task breakdown:
# Stage 1: Move gripper to grasp the tomato sauce (keypoint 0).
# Stage 2: Move the gripper with the grasped tomato sauce to the basket (keypoints 1, 2, 3, 4, 5).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance function for moving the gripper to grasp the tomato sauce (keypoint 0).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Tomato sauce keypoint index
    target_pos = keypoints[target_idx][0]  # Shape (3,)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative squared distance
    return reward.mean()  # Scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No keypoint needs to be selected here for motion to the basket. Just return 0 to indicate no additional guidance is needed.
    return torch.tensor(0.0, device=action_sequence.device)  # Return scalar reward of 0
