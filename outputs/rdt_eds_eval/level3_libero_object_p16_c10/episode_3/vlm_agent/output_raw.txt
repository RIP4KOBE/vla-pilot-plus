# Task breakdown:
# Stage 1: Move gripper to grasp salad dressing (`keypoint 0`) identified from the image
# Stage 2: Place the grasped salad dressing into the basket (`keypoints 2, 3, 4, 5`). No need for guidance function in this stage as per task rules.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move to grasp the salad dressing at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Define the target position as keypoint 0 (salad dressing)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Keypoint 0 corresponds to salad dressing

    # Compute weighted distance to the target
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for this stage as per task-specific rules
    return torch.tensor(0.0, device=action_sequence.device)
