# Task breakdown:
# Stage 1: Grasp the red salad dressing at keypoint 0 or 1 (both correspond to "red_salad_dressing_1_main")
# Stage 2: Place the salad dressing into the basket (keypoints 2, 3, 4, 5, 6)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance for grasping the red salad dressing (keypoint 1)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Emphasizing later part of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Target position: keypoint 1 (red salad dressing)
    target_idx = torch.tensor([1], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Compute distance from trajectory to the target
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decisions needed for placing into the basket; return 0
    return torch.tensor(0.0, device=action_sequence.device)
