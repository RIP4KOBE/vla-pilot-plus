# Task breakdown:
# Stage 1: Move gripper to grasp the alphabet soup (keypoint 0 or keypoint 1) based on the labeled position in the image
# Stage 2: Move gripper with the grasped alphabet soup to the basket (keypoints 2, 3, 4, 5, and 6)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the alphabet soup (keypoint 0 is selected as it represents the correct position of the alphabet soup in the image)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Keypoint 0 corresponds to the alphabet soup.
    target_pos = keypoints[target_idx][0]

    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No need to guide the placement action since the policy knows how to handle placing in the basket.
    return torch.tensor(0.0, device=action_sequence.device)
