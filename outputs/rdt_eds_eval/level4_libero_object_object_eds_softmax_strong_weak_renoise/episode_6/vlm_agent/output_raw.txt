# Task breakdown:
# Stage 1: Move gripper to grasp the tomato sauce (keypoint 0), as identified by its label and location in the image.
# Stage 2: Move gripper with the tomato sauce to the basket (keypoints 1, 2, 3, and 4). No guidance needed here.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Focus the motion toward the keypoint associated with the tomato sauce (keypoint 0).
    
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Final stage has higher weight for accuracy
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()
    
    # Set target position to the tomato sauce keypoint (index 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Compute distances between the action sequence and the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Apply weights and sum over time
    
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance necessary as the robot knows how to place the tomato sauce in the basket.
    return torch.tensor(0.0, device=action_sequence.device)
