def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Goal: Encourage robot to move towards keypoint 0 (butter_1_main)
    T = action_sequence.shape[1]  # Number of timesteps
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Assign higher weights to the later part of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, 3.0, 1.0)
    weights = weights.to(action_sequence.device) / weights.sum()

    # Hardcode keypoint index 0 for butter
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Compute distances between action sequence and target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)

    # Compute weighted reward
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!
