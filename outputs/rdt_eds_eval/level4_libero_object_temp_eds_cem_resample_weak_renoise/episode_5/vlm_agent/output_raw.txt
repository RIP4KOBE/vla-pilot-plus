# Task breakdown:
# Stage 1: Move gripper to grasp the ketchup bottle (keypoint 5), which is identified on the table.
# Stage 2: Move the gripper with the ketchup to the basket (keypoints 0, 1, 2, 3, 4).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance to move the gripper to the ketchup bottle
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Apply higher weights toward the later timesteps
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # The target keypoint is the ketchup bottle at index 5
    target_idx = torch.tensor([5], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape (3,)

    # Calculate 3D distances and reward
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted distances
    return reward.mean()  # Return a scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # There is no decision-making needed for placing the ketchup in the basket.
    # The policy knows how to place it properly, so we simply return 0.
    return torch.tensor(0.0, device=action_sequence.device)
