# Task breakdown:
# Stage 1: Move gripper to grasp the milk (keypoint 4)
# Stage 2: Move gripper with the milk to the basket (keypoints 0, 1, 2, 3)
# Note: No stage 2 guidance function is necessary per task-specific rules since we do not need to make a decision here.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the milk at keypoint 4
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    milk_idx = torch.tensor([4], dtype=torch.long, device=keypoints.device)
    milk_pos = keypoints[milk_idx][0]  # Directly get coordinates of keypoint 4
    
    all_dists = torch.norm(action_sequence - milk_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance is required for placing the object in the basket. Return 0.
    return torch.tensor(0.0, device=action_sequence.device)
