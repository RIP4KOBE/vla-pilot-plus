# Task breakdown:
# Stage 1: Move gripper to grasp the BBQ sauce (keypoint 0).
# Stage 2: Place the grasped object into the basket (keypoints 2-5, no guidance needed for this stage).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Guidance for moving gripper to grasp the BBQ sauce at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # BBQ sauce is at keypoint 0
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Compute distance to target
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative squared distance
    return reward.mean()  # MUST return scalar!

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision-making required for stage 2, as the basket location is predefined
    return torch.tensor(0.0, device=action_sequence.device)  # Return zero reward for stage 2
