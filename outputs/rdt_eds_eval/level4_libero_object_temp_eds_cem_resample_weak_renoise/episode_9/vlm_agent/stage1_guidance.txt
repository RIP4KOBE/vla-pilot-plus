def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the chocolate pudding (keypoints 0 or 1 based on position from image)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Choose keypoint 0 for chocolate pudding
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Apply higher weights near the end of the trajectory to focus on the final grasp position
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()  # Normalize weights

    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Calculate distance for each trajectory point
    reward = -(all_dists**2 * weights).sum(dim=1)  # Negative squared distance weighted by time
    return reward.mean()  # Return scalar reward
