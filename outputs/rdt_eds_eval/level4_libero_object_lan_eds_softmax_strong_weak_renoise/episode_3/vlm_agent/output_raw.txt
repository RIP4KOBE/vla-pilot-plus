# Task breakdown:
# Stage 1: Move gripper to grasp salad dressing (keypoint 0), identified visually from the image.
# Stage 2: Move gripper with salad dressing into the basket (keypoint 2). No specific placement decision needed; policy handles placing.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Stage 1: Guide the gripper to grasp the salad dressing at keypoint 0.
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Salad dressing keypoint
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Compute distances to the salad dressing keypoint
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted reward
    return reward.mean()  # Return a scalar

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Stage 2: No specific placement decision needed for placing into basket.
    # Directly guiding to basket location is sufficient; return 0 as per instructions.
    return torch.tensor(0.0, device=action_sequence.device)
