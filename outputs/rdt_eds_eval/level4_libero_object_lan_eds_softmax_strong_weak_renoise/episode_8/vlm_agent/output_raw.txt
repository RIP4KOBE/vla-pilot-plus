# Task breakdown:
# Stage 1: Move gripper to grasp the milk (keypoint 0)
# Stage 2: Move gripper with milk to the basket (keypoints 2, 3, 4, 5, or 6).
# Note: For stage 2, we do not need to create a guidance reward as the policy knows how to put the milk into the basket.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move the gripper to grasp the milk (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # keypoint for the milk
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Distance to target position shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative squared distance
    return reward.mean()  # Always return scalar

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Move the gripper with milk to the basket (no explicit guidance is needed here)
    return torch.tensor(0.0, device=action_sequence.device)  # Return 0 as the policy automatically handles placing
