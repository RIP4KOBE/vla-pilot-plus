# Task breakdown:
# Stage 1: Move gripper to grasp the butter (keypoint 0), as identified visually in the image.
# Stage 2: Move gripper with the butter into the basket (keypoint 1); no additional guidance needed.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the butter (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)  # Consider increasing weights near the end of the trajectory
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()  # Normalize

    # Target position is the butter keypoint (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Extract position (3,)

    # Compute distances to the butter's position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative squared distance
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No additional decision-making needed for placing; return 0
