# Task breakdown:
# Stage 1: Move gripper to grasp BBQ sauce (keypoint 0), as identified in the image.
# Stage 2: Move gripper with grasped object to basket (keypoints 1, 2, 3, 4, 5). No decision is needed for the basket placement.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Directly guide the robot to grasp the BBQ sauce at keypoint 0.
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Apply higher weight to the later part of the trajectory.
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Define the target position for the BBQ sauce at keypoint 0.
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    
    # Reward function that encourages the action sequence to approach the BBQ sauce keypoint.
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for placing into basket. Basket keypoints are predefined by policy.
    return torch.tensor(0.0, device=action_sequence.device)
