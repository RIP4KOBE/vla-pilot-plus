# Task breakdown:
# Stage 1: Move gripper to grasp the alphabet soup can (keypoint 0).
# Stage 2: Move gripper with the alphabet soup can to the basket (keypoint 1).

num_stages = 2

### Stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance for moving the gripper to grasp the alphabet soup can (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Assign stronger weight toward the end of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Target position: keypoint 0
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Compute distances between the action sequence and the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    
    # Reward is the negative squared distance, weighted over timesteps
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return scalar reward

### Stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Guidance for moving the alphabet soup to the basket (keypoint 1)
    # No need to explicitly guide this motion, return 0 as we let the policy handle placing
    return torch.tensor(0.0, device=action_sequence.device)
