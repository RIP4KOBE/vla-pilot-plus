# Task breakdown:
# Stage 1: Move gripper to grasp the butter (keypoint 0) which is identified visually on the left of the image.
# Stage 2: Move gripper with the butter to place it in the basket (keypoint 1).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the butter (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Target butter position
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Compute the distance to the butter
    distances = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(distances ** 2 * weights).sum(dim=1)
    return reward.mean()  # Ensure scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Stage 2 is moving the butter into the basket (keypoint 1).
    # No specific guidance needed as the placement target is clear.
    return torch.tensor(0.0, device=action_sequence.device)
