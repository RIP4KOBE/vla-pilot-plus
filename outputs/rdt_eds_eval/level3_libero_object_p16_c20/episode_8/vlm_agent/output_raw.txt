# Task breakdown:
# Stage 1: Move gripper to grasp the milk (keypoint 0) which I can see is visually labeled as "milk_1_main".
# Stage 2: Move gripper with milk to the basket (keypoints 1, 2, 3, 4, 5). As the placing decision is not needed, return 0.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the milk at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)  # Focus on final timesteps
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Hardcoded target keypoint for the milk (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Minimize the distance to the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Reward is negative of weighted squared distance
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No need for decision making for the basket placing task; return 0
    return torch.tensor(0.0, device=action_sequence.device)
