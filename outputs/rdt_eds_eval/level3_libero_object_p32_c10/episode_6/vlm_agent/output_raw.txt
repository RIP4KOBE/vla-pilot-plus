# Task breakdown:
# Stage 1: Move gripper to grasp the tomato_sauce (keypoint 0) which I identified visually.
# Stage 2: Move gripper with the grasped tomato_sauce to the basket (keypoints 1, 2, 3, 4, 5).
# Note: For stage 2, no decision is needed since it is just placing into the basket. Directly return 0.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guide the gripper to grasp the target object, tomato_sauce at keypoint 0.
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Target position is tomato_sauce at keypoint 0
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    
    # Calculate distance to target
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return a scalar

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # For placing in the basket, no additional decision-making is needed. Just return 0.
    return torch.tensor(0.0, device=action_sequence.device)
