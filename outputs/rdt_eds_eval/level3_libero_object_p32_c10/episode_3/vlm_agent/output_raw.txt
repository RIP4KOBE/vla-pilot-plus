# Task breakdown:
# Stage 1: Move gripper to grasp the salad dressing (keypoint 0 or 1)
# Stage 2: Move gripper with salad dressing to the basket (keypoints 2, 3, 4, 5)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # The goal here is to guide the robot to the salad dressing (keypoint 0 for higher Z position grasp point).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Weighting prioritizes final trajectories for better precision
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Target is the salad dressing (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]

    # Compute distances between action trajectory and target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Higher reward when closer to target

    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No specific decision is required in stage 2, the placement in the basket is handled by the policy.
    return torch.tensor(0.0, device=action_sequence.device)  # Return static reward 0
