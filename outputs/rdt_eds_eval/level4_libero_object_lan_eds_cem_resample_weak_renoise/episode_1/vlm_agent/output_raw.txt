# Task breakdown:
# Stage 1: Move gripper to grasp the "alphabet soup" (keypoint 0) as identified in the image.
# Stage 2: Move gripper with the grasped "alphabet soup" to the basket (keypoints 1, 2, 3, 4).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to "alphabet soup" (keypoint 0).
    # ALL code must be self-contained here.
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    # Hard-coded index for "alphabet soup" (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # (B, T)
    reward = -(all_dists**2 * weights).sum(dim=1)  # Weighted reward
    return reward.mean()  # Scalar output

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Move gripper to place the object into the basket.
    # NO decision-making required for placing into the basket.
    # The policy already knows where to move.
    return torch.tensor(0.0, device=action_sequence.device)  # Just return 0
