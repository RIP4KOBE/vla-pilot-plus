# Task breakdown:
# Stage 1: Move gripper to grasp orange juice (keypoint 0). The orange juice is visually marked at keypoint 0.
# Stage 2: Move gripper with orange juice to the basket (keypoints 2, 3, 4, 5). No reward function is needed for this stage.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Stage 1: Move gripper to grasp the orange juice (keypoint 0).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    # Applying higher weight to time steps closer to the end of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, 
                          torch.tensor(3.0, device=action_sequence.device), 
                          torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Index for orange juice
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Stage 2: Move gripper with orange juice to the basket
    # No reward function is needed since diffusion policy resolves this placement task on its own.
    return torch.tensor(0.0, device=action_sequence.device)
