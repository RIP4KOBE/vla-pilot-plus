# Task breakdown:
# Stage 1: Move gripper to grasp the milk (keypoint 0), identified visually.
# Stage 2: Move gripper with milk to place it into the basket (keypoints 2, 3, 4, 5, 6). No guidance needed for this stage.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the milk (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Define final section of trajectory where precision is most important
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),  # Heavier weight towards end
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Target is keypoint 0 (milk)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Compute negative square distance to target
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Sum over T, shape: (B,)

    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision needed; the policy knows how to place in the basket
    return torch.tensor(0.0, device=action_sequence.device)
