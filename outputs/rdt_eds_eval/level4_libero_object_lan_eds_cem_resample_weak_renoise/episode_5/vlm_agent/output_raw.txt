# Task breakdown:
# Stage 1: Move gripper to grasp ketchup bottle (keypoint 1), as identified in the image.
# Stage 2: Move gripper with ketchup bottle to the basket (keypoints 2, 3, 4, 5).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the ketchup bottle (keypoint 1)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Hardcoded index for ketchup bottle (keypoint 1)
    target_idx = torch.tensor([1], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Compute reward based on proximity to ketchup bottle
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Negative distance squared for higher reward when closer
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No specific guidance is needed for this stage since the task is to place in the basket
    return torch.tensor(0.0, device=action_sequence.device)  # Policy handles placement
