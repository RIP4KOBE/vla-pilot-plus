# Task breakdown:
# Stage 1: Move gripper to grasp cream cheese (keypoint 0), which I have identified as the target object.
# Stage 2: Move gripper with cream cheese to the basket (keypoint 1, 2, 3, 4). For this task, no guidance function is needed for stage 2 (per instructions).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # The task here is to guide the gripper to grasp the cream cheese at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    # Trajectory weights encourage focus on the final portion of the sequence
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Target position corresponds to keypoint 0 (cream cheese)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Compute distances from the action sequence to the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    
    # Reward function is the negative squared distance, weighted
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return a scalar value

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance needed for stage 2; we just return 0
    return torch.tensor(0.0, device=action_sequence.device)
