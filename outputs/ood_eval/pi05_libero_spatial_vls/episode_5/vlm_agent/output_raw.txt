# Task breakdown:
# Stage 1: Move gripper to grasp the black bowl in the top drawer of the wooden cabinet (keypoint 0).
# Stage 2: Move gripper with the black bowl to the plate (keypoints 6, 7).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the black bowl in the top drawer of the wooden cabinet (keypoint 0).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Keypoint 0 corresponds to the black bowl in the top drawer
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # Move gripper with the black bowl to the plate (keypoints 6, 7).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Use the mean of keypoints 6 and 7 to represent the plate's center
    plate_indices = torch.tensor([6, 7], dtype=torch.long, device=keypoints.device)
    plate_center = keypoints[plate_indices].mean(dim=0)
    all_dists = torch.norm(action_sequence - plate_center, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!
