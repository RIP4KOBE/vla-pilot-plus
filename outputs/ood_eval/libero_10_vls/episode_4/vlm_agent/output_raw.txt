# Task breakdown:
# Stage 1: Move gripper to grasp the black bowl (keypoint 1) which is on the table.
# Stage 2: Move gripper with the black bowl to the bottom drawer of the cabinet (keypoint 10).
# Stage 3: Move gripper to the cabinet handle (keypoint 11) to close the drawer.

num_stages = 3

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the black bowl (keypoint 1)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([1], dtype=torch.long, device=keypoints.device)  # Keypoint 1 for black bowl
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()
