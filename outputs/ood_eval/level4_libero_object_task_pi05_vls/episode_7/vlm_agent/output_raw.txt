# Task breakdown:
# Stage 1: Move gripper to grasp the orange juice (keypoint 5) which is visually identified in the image.
# Stage 2: Move gripper with the orange juice to the basket (keypoint 1). No guidance function is needed for this stage.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Guidance to move the gripper to grasp the orange juice (keypoint 5)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([5], dtype=torch.long, device=keypoints.device)  # Keypoint for orange juice
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative squared distance
    return reward.mean()  # MUST return scalar!
