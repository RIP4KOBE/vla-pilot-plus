# Task breakdown:
# Stage 1: Move gripper to grasp chocolate pudding (keypoint 0 or 1)
# Stage 2: Move gripper with chocolate pudding to the basket (keypoints 2, 3, 4, 5)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp chocolate pudding (keypoint 0 or 1)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    # Target keypoint for chocolate pudding
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Keypoint 0 corresponds to chocolate pudding
    target_pos = keypoints[target_idx][0]  # Shape: (3,)
    
    # Compute distance to target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted negative squared distance
    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No decision needed for placing in the basket, just return 0
    return torch.tensor(0.0, device=action_sequence.device)
