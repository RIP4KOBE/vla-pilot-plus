# Task breakdown:
# Stage 1: Move gripper to grasp cream cheese (keypoint 0), which is identified by looking at the image.
# Stage 2: Move the grasped cream cheese to the basket (keypoints 1, 2, 3, 4), which is identified as the basket by visually inspecting the image.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guide the robot to move the gripper to the cream cheese (keypoint 0).
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    cream_cheese_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    cream_cheese_pos = keypoints[cream_cheese_idx][0]  # Shape: (3,)
    all_dists = torch.norm(action_sequence - cream_cheese_pos, dim=-1)  # Distance to cream cheese
    reward = -(all_dists**2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No need to provide a guidance function here, as placing into the basket does not require a decision.
    return torch.tensor(0.0, device=action_sequence.device)
