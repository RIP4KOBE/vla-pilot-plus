# Task breakdown:
# Stage 1: Move gripper to grasp BBQ sauce (keypoint 0), which is visually identified in the image.
# Stage 2: Place the grasped BBQ sauce into the basket (keypoints 1, 2, 3, 4, 5), which represents the basket area. 

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to the position of BBQ sauce at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # BBQ sauce keypoint index
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!

    # No need to create a reward for this stage as basket placement requires no decision-making.
