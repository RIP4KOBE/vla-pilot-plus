# Task breakdown:
# Stage 1: Move gripper to grasp the orange juice (keypoint 1).
# Stage 2: Move gripper with orange juice to the basket (keypoints 2, 3, 4, 5, 6).

num_stages = 2

### Stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)

    # Assigning higher reward to later timesteps
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    # Target is keypoint 1 (orange juice)
    target_idx = torch.tensor([1], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Compute negative squared distance to the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return a scalar reward

### Stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # We do not need to guide this stage; the policy knows how to place into the basket.
    return torch.tensor(0.0, device=action_sequence.device)
