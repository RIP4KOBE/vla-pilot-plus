# Task breakdown:
# Stage 1: Move gripper to grasp the milk (keypoint 0)
# Stage 2: Move gripper with milk to the basket (keypoints 1, 2, 3, 4, 5)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Moving the gripper to grasp the milk
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)  # Last 30% gets higher weight
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)  # Index for milk
    target_pos = keypoints[target_idx][0]  # Get position of keypoint 0
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # 3D distances
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Negative squared distance weighted
    return reward.mean()  # MUST return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance required as the robot knows how to place correctly in the basket
    return torch.tensor(0.0, device=action_sequence.device)
