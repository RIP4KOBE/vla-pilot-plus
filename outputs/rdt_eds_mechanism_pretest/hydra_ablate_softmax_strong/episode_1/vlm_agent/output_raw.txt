# Task breakdown:
# Stage 1: Move gripper to grasp the cream cheese (keypoint 0).
# Stage 2: Place the grasped cream cheese in the basket (keypoints 1, 2, 3, 4, 5). No detailed reward guidance is needed; policy already knows where to place.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move gripper to grasp the cream cheese at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)

    # Assign higher weight for the final part of the trajectory
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()

    # Extract the target keypoint for the cream cheese
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]

    # Compute distances and the reward
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists**2 * weights).sum(dim=1)
    return reward.mean()  # Return a scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No detailed reward guidance is needed for placing in the basket
    return torch.tensor(0.0, device=action_sequence.device)
