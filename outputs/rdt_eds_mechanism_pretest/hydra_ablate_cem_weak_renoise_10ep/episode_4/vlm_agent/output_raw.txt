# Task breakdown:
# Stage 1: Move gripper to grasp the BBQ sauce (keypoint 0), identified visually from the image.
# Stage 2: Move gripper with the BBQ sauce to place it in the basket (keypoints 1, 2, 3, 4, 5).

num_stages = 2

### stage 1 guidance (grasp BBQ sauce)
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    
    # Emphasize the pick motion to the BBQ sauce (keypoint 0)
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    
    # Assign higher weights to the last part of the trajectory
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()
    
    # Target the BBQ sauce keypoint
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape (3,)
    
    # Compute the distances and return the weighted reward
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape (B, T)
    reward = -(all_dists**2 * weights).sum(dim=1)  # Reward higher for closer
    return reward.mean()  # MUST return scalar!

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No specific decision needed for placing in the basket - directly return 0
    return torch.tensor(0.0, device=action_sequence.device)
