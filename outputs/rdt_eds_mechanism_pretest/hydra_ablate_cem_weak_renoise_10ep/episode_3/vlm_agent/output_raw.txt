# Stage 2: Move gripper with salad dressing to the basket (keypoints 2, 3, 4, 5), which I can see is on the right side of the scene. No decision needed for placement - policy handles this.

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Guidance for moving the gripper to grasp the salad dressing at keypoint 0.

    # Define time-dependent weights
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device),
    )
    weights = weights / weights.sum()  # Normalize weights
    
    # Target keypoint corresponding to the salad dressing (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    
    # Compute distances between action sequence and target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # MUST return scalar!
