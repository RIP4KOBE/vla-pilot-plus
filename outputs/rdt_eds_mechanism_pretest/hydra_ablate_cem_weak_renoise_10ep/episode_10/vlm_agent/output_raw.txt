# Task breakdown:
# Stage 1: Move gripper to grasp the orange juice (keypoint 0)
# Stage 2: Move gripper with orange juice to the basket (no guidance needed)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance to move the gripper to grasp the orange juice at keypoint 0
    T = action_sequence.shape[1]
    
    # Set final trajectory weights emphasizing later time steps
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(timesteps >= final_start, torch.tensor(3.0, device=action_sequence.device), torch.tensor(1.0, device=action_sequence.device))
    weights = weights / weights.sum()  # Normalize weights

    # Identify the target position for the orange juice (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]  # Shape: (3,)

    # Compute the distance between the action sequence and the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Shape: (B, T)
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted sum of distances

    return reward.mean()  # Return scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No guidance is needed since the action is to move to the basket
    return torch.tensor(0.0, device=action_sequence.device)
