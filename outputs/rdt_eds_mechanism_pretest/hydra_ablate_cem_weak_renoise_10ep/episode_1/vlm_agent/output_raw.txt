# Task breakdown:
# Stage 1: Move gripper to grasp the alphabet soup (keypoint 0)
# Stage 2: Move the grasped alphabet soup to the basket (no decision needed, guidance returns 0)

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Guidance for reaching the alphabet soup at keypoint 0
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)  # Final portion of trajectory
    weights = torch.where(
        timesteps >= final_start,
        torch.tensor(3.0, device=action_sequence.device),
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)  # Compute distances
    reward = -(all_dists ** 2 * weights).sum(dim=1)  # Weighted sum of squared distances
    return reward.mean()  # Return final scalar reward

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No specific guidance is needed for placing the alphabet soup into the basket
    # The policy automatically handles this action
    return torch.tensor(0.0, device=action_sequence.device)
