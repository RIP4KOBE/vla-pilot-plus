# Task breakdown:
# Stage 1: Move gripper to grasp the chocolate pudding (keypoint 0), which is clearly labeled as chocolate_pudding_1_main.
# Stage 2: Move gripper with the chocolate pudding to the basket (keypoints 1, 2, 3, 4). No specific guidance is needed here; just return 0.

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # ALL code must be inside this function - no external helpers!
    # Guidance for moving the gripper to the chocolate pudding at keypoint 0.
    
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    
    # Higher reward towards the end of the trajectory
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Target is the keypoint for the chocolate pudding
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]
    
    # Compute the distance between the end-effector and the target location
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    
    return reward.mean()  # Return a scalar to guide the robot.

### stage 2 guidance
def stage2_guidance(keypoints, action_sequence):
    # No specific decision is needed here since the policy knows how to put objects into the basket.
    return torch.tensor(0.0, device=action_sequence.device)
