# Task breakdown:
# Stage 1: Move gripper to grasp the tomato sauce (keypoint 0).
# Stage 2: Move gripper with tomato sauce to the basket (we do not need to create a specific reward for placing into the basket as per the task's rules).

num_stages = 2

### stage 1 guidance
def stage1_guidance(keypoints, action_sequence):
    # Move to grasp the tomato sauce.
    T = action_sequence.shape[1]
    timesteps = torch.arange(T, device=action_sequence.device)
    final_start = int(0.7 * T)
    weights = torch.where(
        timesteps >= final_start, 
        torch.tensor(3.0, device=action_sequence.device), 
        torch.tensor(1.0, device=action_sequence.device)
    )
    weights = weights / weights.sum()

    # Keypoint for tomato sauce (keypoint 0)
    target_idx = torch.tensor([0], dtype=torch.long, device=keypoints.device)
    target_pos = keypoints[target_idx][0]

    # Calculate distance to the target position
    all_dists = torch.norm(action_sequence - target_pos, dim=-1)

    # Reward is negative squared distance weighted by time
    reward = -(all_dists ** 2 * weights).sum(dim=1)
    return reward.mean()  # Return a scalar reward

    # We do not need to create reward for placing into basket as per the task's rules.
